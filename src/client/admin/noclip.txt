import { toogleAnim } from '../player/toogleAnim.js';

let enabledNoclip = false;
let speed = 1.0;
let everyTick = null;

const KEYS = {
  W: 32,
  S: 33,
  A: 34,
  D: 35,
  SPACE: 22,
  CTRL: 36,
  SHIFT: 21,
};

alt.on('keyup', (key) => {
  if (key === 116) { // F5
    toggleNoclip();
  }
});

const toggleNoclip = () => {
  enabledNoclip = !enabledNoclip;

  if (enabledNoclip) {
    disableControls()
    toogleAnim(true, 'anim@amb@business@meth@meth_smash_weight_check@', 'break_weigh_v2_methbag01^4', -1)
    native.freezeEntityPosition(alt.Player.local.scriptID, true);
    native.setEntityAlpha(alt.Player.local.scriptID, 50, false)
    everyTick = alt.everyTick(keyHandler);
    alt.log('Noclip включен');
  } else {
    enableControls()
    toogleAnim(false, 'anim@amb@business@meth@meth_smash_weight_check@', 'break_weigh_v2_methbag01^4', -1)
    native.freezeEntityPosition(alt.Player.local.scriptID, false);
    native.setEntityAlpha(alt.Player.local.scriptID, 255, false)
    alt.clearEveryTick(everyTick);
    alt.log('Noclip выключен');
  }
};

const keyHandler = () => {
  let currentSpeed = speed;
  let pos = alt.Player.local.pos; // Получаем текущую позицию
  let rot = native.getGameplayCamRot(2);
  let dirForward = camVectorForward(rot);
  let dirRight = camVectorRight(rot);

  if (native.isDisabledControlPressed(0, KEYS.SHIFT)) {
    currentSpeed *= 5; // Увеличиваем скорость при нажатом Shift
  }

  // Обновляем позицию в зависимости от нажатых клавиш
  if (native.isDisabledControlPressed(0, KEYS.W)) {
    pos = addSpeedToVector(pos, dirForward, currentSpeed);
  }

  if (native.isDisabledControlPressed(0, KEYS.S)) {
    pos = addSpeedToVector(pos, dirForward, -currentSpeed);
  }

  if (native.isDisabledControlPressed(0, KEYS.A)) {
    pos = addSpeedToVector(pos, dirRight, -currentSpeed);
  }

  if (native.isDisabledControlPressed(0, KEYS.D)) {
    pos = addSpeedToVector(pos, dirRight, currentSpeed);
  }

  let zMod = 0;

  if (native.isDisabledControlPressed(0, KEYS.SPACE)) {
    zMod += currentSpeed;
  }

  if (native.isDisabledControlPressed(0, KEYS.CTRL)) {
    zMod -= currentSpeed;
  }

  // Создаем новый вектор с обновленной высотой
  const newPos = new alt.Vector3(pos.x, pos.y, pos.z + zMod);

  if (!isVectorEqual(newPos, alt.Player.local.pos)) {
    alt.emitServer('client:noclip:setPos', newPos.x, newPos.y, newPos.z);
  }

  // Поворачиваем персонажа в сторону камеры
  let heading = (native.getGameplayCamRot(2).z) % 360;
  native.setEntityHeading(alt.Player.local.scriptID, heading);
  
  if (enabledNoclip) {
    if (!native.isEntityPlayingAnim(alt.Player.local.scriptID, 'anim@amb@business@meth@meth_smash_weight_check@', 'break_weigh_v2_methbag01^4', 3)) {
      native.taskPlayAnim(alt.Player.local.scriptID, 'anim@amb@business@meth@meth_smash_weight_check@', 'break_weigh_v2_methbag01^4', 1, -1, -1, 1, 1, false, false, false);
    }
  }
};

const addSpeedToVector = (vector1, vector2, speed, lr = false) => {
  return new alt.Vector3(
    vector1.x + vector2.x * speed,
    vector1.y + vector2.y * speed,
    lr ? vector1.z : vector1.z + vector2.z * speed
  );
};

const camVectorForward = (camRot) => {
  let rotInRad = {
    x: camRot.x * (Math.PI / 180),
    y: camRot.y * (Math.PI / 180),
    z: camRot.z * (Math.PI / 180),
  };

  return {
    x: -Math.sin(rotInRad.z) * Math.cos(rotInRad.x), // Учитываем угол по X
    y: Math.cos(rotInRad.z) * Math.cos(rotInRad.x),  // Учитываем угол по X
    z: Math.sin(rotInRad.x) // Учитываем угол по Y (вверх/вниз)
  };
};

const camVectorRight = (camRot) => {
  let rotInRad = {
    x: camRot.x * (Math.PI / 180),
    y: camRot.y * (Math.PI / 180),
    z: camRot.z * (Math.PI / 180),
  };

  return {
    x: Math.cos(rotInRad.z),  // Направление вправо
    y: Math.sin(rotInRad.z),  // Направление вправо
    z: 0 // Высота не меняется при движении влево/вправо
  };
};
const isVectorEqual = (vector1, vector2) => {
  return (
    vector1.x === vector2.x &&
    vector1.y === vector2.y &&
    vector1.z === vector2.z
  );
};

const disableControls = () => {
  native.disableControlAction(0, 0, true);   // Look Left/Right
  native.disableControlAction(0, 1, true);   // Look Up/Down
  native.disableControlAction(0, 2, true);   // Move Up/Down
  native.disableControlAction(0, 14, true);  // Attack
  native.disableControlAction(0, 15, true);  // Aim
  native.disableControlAction(0, 24, true);  // Jump
  native.disableControlAction(0, 25, true);  // Enter Vehicle
  native.disableControlAction(0, 30, true);  // Move Left/Right
  native.disableControlAction(0, 31, true);  // Move Forward/Backward
  native.disableControlAction(0, 32, true);  // Move Up/Down
  native.disableControlAction(0, 33, true);  // Move Left/Right
  native.disableControlAction(0, 34, true);  // Move Forward/Backward
  native.disableControlAction(0, 49, true);  // Look Behind
  native.disableControlAction(0, 37, true);  // Select Weapon
  native.disableControlAction(0, 53, true);  // Cover
  native.disableControlAction(0, 54, true);  // Reload
  native.disableControlAction(0, 56, true);  // Use
  native.disableControlAction(0, 157, true); // Special Ability
  native.disableControlAction(0, 158, true); // Special Ability 2
  native.disableControlAction(0, 159, true); // Special Ability 3
  native.disableControlAction(0, 160, true); // Special Ability 4
  native.disableControlAction(0, 161, true); // Special Ability 5
  native.disableControlAction(0, 162, true); // Special Ability 6
  native.disableControlAction(0, 163, true); // Special Ability 7
  native.disableControlAction(0, 164, true); // Special Ability 8
  native.disableControlAction(0, 165, true); // Special Ability 9
};

const enableControls = () => {
  for (let i = 0; i <= 165; i++) {
    native.enableControlAction(0, i, true); // Включаем все действия
  }
};
